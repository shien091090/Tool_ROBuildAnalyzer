def collect_missing(expr: str, variables: dict, ctx: CalcContext, current_slot: int | None) -> set[str]:
    """只回報expr裡解不開的角色資料key(skill:N / get:N / SCALAR_KEYS), 不eval。
    給parser判斷「這行算不出來是因為缺角色資料, 還是真的看不懂」用。"""
    _normalized, missing = normalize(expr, variables, ctx, current_slot)
    return set(missing)

```python
_RE_IDENT = re.compile(r"[A-Za-z_]\w*")


def _line_missing(text, variables, ctx, slot_id, unresolved_vars):
    """text(整行或右式)缺的角色資料 = 直接出現的GetSkillLevel/get/SCALAR key
    ∪ 引用到的未解變數所缺的key。via列出經由哪些變數帶進來(給condition文字)。"""
    missing = lua_expr.collect_missing(text, variables, ctx, slot_id)
    via: list[str] = []
    for ident in sorted(set(_RE_IDENT.findall(text))):
        if ident in unresolved_vars:
            var_missing, rhs = unresolved_vars[ident]
            missing |= var_missing
            via.append(f"{ident} = {rhs}")
    return missing, via


def _unresolved_line_entry(line: str, missing: set[str], via: list[str]) -> EffectEntry:
    condition = ("依賴變數 " + "; ".join(via)) if via else ("依賴角色資料 " + ", ".join(sorted(missing)))
    return EffectEntry(
        key="未判定條件", value=None, unit="", kind=KIND_UNRESOLVED, category=CAT_OTHER,
        extra={"condition": condition, "missing": sorted(missing), "raw_lines": [line]},
    )
```

def _strip_junk(value, junk):
    for j in junk:

def _expand_positions(raw):
    raw = _strip_junk(raw, _POSITION_JUNK)
    if not raw:

    if raw.startswith("頭"):
        body = raw[1:].replace("/", "")

_CARD_SIDE_EXCLUDE = {"acc_r": "(左)", "acc_l": "(右)"}
_STONE_SUFFIX = {"costume_top": "(頭上)", "costume_mid": "(頭中)", "costume_low": "(頭下)", "costume_garment": "(披肩)"}


class ItemCatalog:
    def __init__(self, entries: Iterable[CatalogEntry]):
        self._by_id = {e.item_id: e for e in entries}
        self._by_name = sorted(self._by_id.values(), key=lambda e: e.display_name)

    @classmethod
    def from_reader(cls, reader: DbReader) -> "ItemCatalog":
        entries = []
        for item_id, internal, display, desc, slot_count, equip_type, has_onstart in reader.iter_catalog_rows():
            if not display:
                continue
            series, positions, equip_text = parse_description_fields(desc)
            entries.append(CatalogEntry(item_id, internal, display, desc or "", slot_count or 0, equip_type,
                                        has_onstart, series, positions, equip_text))
        return cls(entries)

    def get(self, item_id):
        return self._by_id.get(item_id)

    @staticmethod
    def _is_weapon(e):
        return e.equip_type in _WEAPON_TYPES

    @staticmethod
    def _equip_matches(e, slot_key):
        rule = _EQUIP_RULES.get(slot_key)
        if rule is None or e.equip_type != "armor":
            return False
        series_ok, positions_req = rule
        if e.series not in series_ok:
            return False
        return positions_req is None or bool(e.positions & positions_req)

    def equipment_for_slot(self, slot_key, job_id=None):
        if slot_key == "weapon":
            out = [e for e in self._by_name if self._is_weapon(e)]
        else:
            out = [e for e in self._by_name if self._equip_matches(e, slot_key)]
        if job_id is not None:
            out = [e for e in out if jobs.is_equippable_by(job_id, e.equip_text)]
        return out

    def cards_for_slot(self, slot_key):
        rule = _CARD_RULES.get(slot_key)
        if rule is None:
            return []
        exclude = _CARD_SIDE_EXCLUDE.get(slot_key)
        out = []
        for e in self._by_name:
            if e.equip_type != "card" or e.series != "卡片" or not e.equip_text:
                continue
            if "所有欄位皆可" in e.equip_text:
                out.append(e)
                continue
            if any(k in e.equip_text for k in rule) and not (exclude and exclude in e.equip_text):
                out.append(e)
        return out

    def stones_for_slot(self, slot_key):
        suffix = _STONE_SUFFIX.get(slot_key)
        if suffix is None:
            return []
        return [e for e in self._by_name
                if e.equip_type == "card" and e.has_onstart and e.display_name.endswith(suffix)]

    def unclassified_count(self) -> int:
        count = 0
        for e in self._by_name:
            if e.equip_type == "armor" and not any(self._equip_matches(e, k) for k in _EQUIP_RULES):
                count += 1
        return count

    @staticmethod
    def search(entries, query):
        q = (query or "").strip()
        return list(entries) if not q else [e for e in entries if q in e.display_name]

def build_to_dict(build: Build) -> dict:
    slots: dict[str, dict] = {}
    for slot_key in SLOT_IDS:
        cfg = build.slots.get(slot_key)
        if cfg is None:
            continue
        d: dict = {"item_id": cfg.item_id}
        if cfg.refine:
            d["refine"] = cfg.refine
        if cfg.grade != "none":
            d["grade"] = cfg.grade
        if cfg.cards:
            d["cards"] = list(cfg.cards)
        enchants = list(cfg.enchants)
        while enchants and enchants[-1] is None:
            enchants.pop()
        if enchants:
            d["enchants"] = enchants
        if cfg.cost_targets is not None:
            ct = cfg.cost_targets
            ctd: dict = {"refine_from": ct.refine_from, "grade_from": ct.grade_from, "enchant_strategy": ct.enchant_strategy}
            if ct.refine_table is not None:
                ctd["refine_table"] = ct.refine_table
            if ct.enchant_goal is not None:
                ctd["enchant_goal"] = [ct.enchant_goal[0], ct.enchant_goal[1]]
            d["cost_targets"] = ctd
        slots[slot_key] = d
    return {"name": build.name, "slots": slots}

def resolve_db_path(config_path: str = "config.json") -> str:
    """config.json 的 db_path 欄位(不存在/不合法時退回預設 data/ro_items.db)."""
    try:
        return load_config(config_path).db_path
    except (ConfigNotFoundError, ConfigInvalidError):

@dataclass(frozen=True)
class EnchantOption:
    option_internal_name: str
    display_name: str
    weight: int
    probability: Fraction
    description: str


@dataclass(frozen=True)
class EnchantSlotPool:
    slot_index: int

class AnalyzerSession:
    def __init__(self, db_path, *, rules_path="userdata/refine_rules.json", prices_path="userdata/prices.json",
                 manual_path="userdata/manual_enchants.json", builds_dir="userdata/builds",
                 characters_dir="userdata/characters"):
        self.db_path = db_path
        self.reader = DbReader(db_path)
        self.maps: EffectMaps = make_maps(db_path)
        self.catalog = ItemCatalog.from_reader(self.reader)
        self.rules: CostRules = load_rules(rules_path)
        self.prices: dict[str, int] = load_prices(prices_path)
        self.manual: dict = json.loads(Path(manual_path).read_text(encoding="utf-8"))
        self.builds_dir = Path(builds_dir)
        self.characters_dir = Path(characters_dir)

    @classmethod
    def from_config(cls, config_path="config.json"):
        return cls(resolve_db_path(config_path))

    def close(self):
        self.reader.close()

    def import_date(self):
        return self.reader.meta("import_date")

    def list_builds(self):
        return sorted(self.builds_dir.glob("*.json")) if self.builds_dir.exists() else []

    def list_characters(self):
        return sorted(self.characters_dir.glob("*.json")) if self.characters_dir.exists() else []

    def load_build(self, path):
        return load_build(path)

    def save_build(self, build, path):
        save_build(build, path)

    def load_character(self, path):
        return load_character(path)

    def evaluate(self, build: Build, character: Character) -> BuildEffects:
        return evaluate_build(build, character, self.reader, self.maps)

    def compare(self, a, b) -> list[CompareRow]:
        return compare_builds(a, b)

    def cost(self, build: Build) -> BuildCostReport:
        return evaluate_build_cost(build, self.rules, self.prices, self.reader, self.manual)

    def enchant_pools(self, item_id: int) -> list[EnchantSlotPool]:
        entry = self.catalog.get(item_id)
        if entry is None or not entry.internal_name:
            return []
        found = rows_for_item(self.reader, self.manual, entry.internal_name)
        if found is None:
            return []
        _table_key, rows = found  # rows: (slot_index, require_cost, option_internal_name, option_weight)
        by_slot: dict[int, dict[str, int]] = {}
        for slot_index, _rc, option, weight in rows:
            slot = by_slot.setdefault(slot_index, {})
            slot[option] = slot.get(option, 0) + weight  # 重複列權重相加=與spec §7.5分母一致
        pools = []
        for slot_index in sorted(by_slot, reverse=True):
            total = sum(by_slot[slot_index].values())
            options = []
            for option, weight in sorted(by_slot[slot_index].items(), key=lambda kv: (-kv[1], kv[0])):
                rec = self.reader.item_by_internal_name(option)
                display = rec.display_name if rec and rec.display_name else option
                options.append(EnchantOption(option, display, weight,
                                             Fraction(weight, total) if total else Fraction(0),
                                             rec.description if rec else ""))
            pools.append(EnchantSlotPool(slot_index, tuple(options)))
        return pools

    def refine_table_choices(self):
        return sorted((k, self.rules.table_displays.get(k, k)) for k in self.rules.refine_tables)

        return rec.display_name if rec and rec.display_name else internal_name

    def describe_missing(self, missing):
        return describe_missing(missing, self.maps.skill_map)

import html
import re

_RE_COLOR = re.compile(r"\^([0-9A-Fa-f]{6})")


def description_to_html(description):
    if not description:
        return ""
    out = []
    open_span = False
    pos = 0
    for m in _RE_COLOR.finditer(description):
        out.append(html.escape(description[pos:m.start()]))
        if open_span:
            out.append("</span>")
            open_span = False
        code = m.group(1).lower()
        if code != "000000":
            out.append(f'<span style="color:#{code}">')
            open_span = True
        pos = m.end()
    out.append(html.escape(description[pos:]))
    if open_span:
        out.append("</span>")
    return "".join(out).replace("\n", "<br>")

    def active_combo_members(self, build: Build, item_id: int) -> set[int]:
        equipped = set()
        for cfg in build.slots.values():
            equipped.add(cfg.item_id)
            equipped.update(cfg.cards)
        rec = self.reader.item(item_id)
        if rec is None or not rec.combi_ids:

        members_all: set[int] = set()
        for combo_id in rec.combi_ids:

            if combo is None:
                continue
            members, _src = combo
            if members and set(members).issubset(equipped):
